    id = hosttool.1000
    title = "Move Host Tool Confirmation"
    desc = hosttool.1000.desc
    picture = GFX_report_event_001
    is_triggered_only = yes
    
    option = {
        name = "Yes, move the host tool"
        # Store new host globally BEFORE clearing flags
        set_variable = { global.new_host_country = temp_new_host }
        # Clear host flag from everyone
        every_country = {
            clr_country_flag = host
            clr_country_flag = pause_blocked
            clr_country_flag = transfer_enabled
        }
        # Set new host
        var:temp_new_host = { 
            set_country_flag = host 
            set_country_flag = transfer_enabled
        }
        # Send ONE global notification
        news_event = { id = hosttool.1001 }
        clear_variable = temp_new_host
    }
    
    option = {
        name = "No, cancel"
        clear_variable = temp_new_host
    }
}

# Annex confirmation (fired to the host; vars target/taker were set by annex_country_click)
country_event = {
    id = hosttool.63
    title = "Annex Confirmation"
    desc = hosttool.63.desc
    picture = GFX_report_event_001
    is_triggered_only = yes

    option = {
        name = "Yes, annex them"
        var:ROOT.target = {
            var:ROOT.taker = {
                annex_country = { target = PREV transfer_troops = yes }
                every_state = {
                    limit = {
                        is_owned_by = var:ROOT.taker
                        is_core_of = var:ROOT.target
                    }
                    add_compliance = 80
                }
            }
        }
        set_variable = { global.ht_diplo_a = ROOT.target }
        set_variable = { global.ht_diplo_b = ROOT.taker }
        every_country = { limit = { is_ai = no } news_event = { id = hosttool.64 } }
        for_each_scope_loop = {
            array = marked_states
            clear_variable = mark_status
        }
        clear_array = marked_states
        clear_variable = target
        clear_variable = taker
        clear_variable = first_marked_state
    }

    option = {
        name = "No, cancel"
        clear_variable = target
        clear_variable = taker
    }
}
