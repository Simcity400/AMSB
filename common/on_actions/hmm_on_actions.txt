	on_unit_leader_created = { # More performant than only adding no_grind through monthly checks
		effect = {
			if = {
				limit = {
					is_army_leader = yes
					NOT = { has_trait = no_grind }
					FROM = { can_grind_generals = no }
				}
				add_unit_leader_trait = no_grind
			}
		}

	on_monthly = {
		effect = {

				}
				every_country = {
					limit = {
						axis = yes
						
					}
					every_army_leader = {
						remove_unit_leader_trait = no_grind
					}
				}
				set_global_flag = grind_allowed
			}
			every_country = {
				limit = {
					is_playable_country = yes
					NOT = {
						OR = {
							has_war = yes
							has_idea = peace_training
							AND = {
								tag = SPR
								any_country = {
									has_volunteers_amount_from = {
										tag = SPR
										count > 0
									} 
								}	
							}
						}
					}
				}
				add_ideas = peace_training
			}
			if = {
				limit = {
					can_grind_generals = no

				}
				every_army_leader = {
					limit = { NOT = { has_trait = no_grind } }
					add_unit_leader_trait = no_grind
				}
			}
		}
	}

	on_state_control_changed = {
		effect = {
			if = {
				limit = {
					FROM = { tag = FRA }

					OR ={  AND ={  FRA = { surrender_progress > 0.32 NOT = { controls_state = 16 } } }	FRA = { surrender_progress	> 0.5	}	 }

				}


			FRA = { 
				set_country_flag = france_tp
				transfer_units_fraction = {
					target = IRE
					size = 1
					stockpile_ratio = 1
					army_ratio = 0
					navy_ratio = 0
					air_ratio = 0
					keep_unit_leaders_trigger = {
						always = yes
						}
					}
					add_offsite_building = { # From COG being gone
						type = industrial_complex
						level = 4
					}
					add_offsite_building = {
						type = arms_factory 
						level = 3
					}					
				}
				
			}

			#VP System

			if = {
				limit = {
					FROM.FROM = {
						OR = {
							state = 613 #Nanjing
							state = 439 #Dehli
							state = 447 #Alexandria
							state = 450 #Benghasi
							state = 7 #Amsterdam
							state = 16 #Paris
							state = 20 #Lyon
							state = 224 #Bryansk
							state = 219 #Moscow
							state = 217 #Stalingrad
						}
					}
				}
				ROOT = { #gained controll (add var)
					if = {
						limit = {
							allies = yes
						}
						add_to_variable = {
							var = global.allies_score
							value = 1
						}
						clamp_variable = {
							var = global.allies_score
							min = 0
							max = 10
						}
					}
					else_if = {
						limit = { 
							axis = yes
						}
						add_to_variable = {
							var = global.axis_score
							value = 1
						}
						clamp_variable = {
							var = global.axis_score
							min = 0
							max = 10
						}
					}
				}
				FROM = { #lost controll (sub var)
					if = {
						limit = {
							allies = yes
						}
						subtract_from_variable = {
							var = global.allies_score
							value = 1
						}
						clamp_variable = {
							var = global.allies_score
							min = 0
							max = 10
						}
					}
					else_if = {
						limit = { 
							axis = yes
						}
						subtract_from_variable = {
							var = global.axis_score
							value = 1
						}
						clamp_variable = {
							var = global.axis_score
							min = 0
							max = 10
						}
					}
				}
			}

		}
	}
