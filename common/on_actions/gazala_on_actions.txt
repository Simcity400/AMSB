# =============================================================================
# WAR OF GAZALA - desert-war attrition mechanic (restored & fixed 2026-06-21)
# =============================================================================
# When the North African war opens (Italy at war with the UK or France) the
# whole African theatre takes a heavy combat penalty for BOTH sides, simulating
# the supply-starved desert seesaw. The penalty then ramps linearly back to 0
# over ~6 months and removes itself.
#
# Modifier definitions: common/dynamic_modifiers/gazala_dynamic_modifiers.txt
#
# WHY on_monthly_WST (and not a plain on_monthly):
#   on_monthly fires once PER COUNTRY. The original ran the decay there, so the
#   +step was applied ~60x/month and the debuff resolved in days, not months -
#   and that per-country rapid-stepping is what let the value skip past 0.
#   on_monthly_WST fires once (only for the always-present wasteland tag), so the
#   decay is a true once-a-month tick. Root scope is WST; everything below uses
#   global vars / explicit ITA scope / every_state, so root is irrelevant.
#
# THE OLD BUG ("ticked positive instead of stopping at 0"):
#   cleanup was gated on an EXACT-equality check (var = 0). HOI4 stores script
#   variables in fixed-point, so repeated +step accumulation steps OVER 0 without
#   ever landing on it -> cleanup never fired -> the value stuck at a small
#   positive (a permanent +attack BUFF). Fixed with a >= 0 clamp (see below).
#
# ACTIVATION runs in on_daily_WST (2026-07-23): it used to sit in the monthly
# tick, which delayed the modifier by up to a month after Italy's declaration
# (and a full month on 1941-save loads). Daily costs nothing (one tag, a flag
# check that short-circuits forever once fired) and starts the penalty at most
# one day after the war opens. DECAY stays monthly. Because activation can now
# land mid-month, the first monthly decay tick after activation is SKIPPED
# (global.war_of_gazala_skip_first_decay) so the theatre always gets a full
# month at -50% / -25% regardless of tick ordering.

on_actions = {

	on_monthly_WST = {
		effect = {

			# ---------------------------------------------------------------
			# DECAY (monthly). Ramp both penalties toward 0, then tear the whole
			# mechanic down. Steps chosen so attack (-0.5) and speed (-0.25) both
			# cross 0 on the 6th decay tick: -0.5 + 6*0.084 = +0.004 and
			# -0.25 + 6*0.042 = +0.002, so both resolve and are removed together.
			# ---------------------------------------------------------------
			if = {
				limit = { has_global_flag = war_of_gazala_active }

				# First monthly tick after a mid-month activation: consume the
				# skip token instead of decaying, so month 1 stays at full force.
				if = {
					limit = { check_variable = { global.war_of_gazala_skip_first_decay > 0 } }
					clear_variable = global.war_of_gazala_skip_first_decay
				}
				else = {
					add_to_variable = { var = global.war_of_gazala_modifier_attack_factor_var value = 0.084 }
					add_to_variable = { var = global.war_of_gazala_modifier_speed_factor_var value = 0.042 }

					# Fixed-point safety: never let speed overshoot into a positive buff.
					if = {
						limit = { check_variable = { global.war_of_gazala_modifier_speed_factor_var > 0 } }
						set_variable = { var = global.war_of_gazala_modifier_speed_factor_var value = 0 }
					}
				}

				# Save compatibility: Matruh was missing from the original theatre
				# list, so already-active saves need it flagged and modifiered once.
				every_state = {
					limit = {
						state = 452 # Marsa Matruh
						NOT = { has_state_flag = war_in_africa }
					}
					set_state_flag = war_in_africa
					add_dynamic_modifier = { modifier = war_of_gazala_modifier scope = ITA }
					add_dynamic_modifier = { modifier = war_of_gazala_modifier scope = GER }
					add_dynamic_modifier = { modifier = war_of_gazala_modifier scope = HUN }
					add_dynamic_modifier = { modifier = war_of_gazala_modifier scope = BUL }
					add_dynamic_modifier = { modifier = war_of_gazala_modifier scope = ROM }
					add_dynamic_modifier = { modifier = war_of_gazala_modifier scope = FIN }
					add_dynamic_modifier = { modifier = war_of_gazala_modifier scope = SPR }
					add_dynamic_modifier = { modifier = war_of_gazala_modifier scope = ENG }
					add_dynamic_modifier = { modifier = war_of_gazala_modifier scope = FRA }
					add_dynamic_modifier = { modifier = war_of_gazala_modifier scope = SAF }
					add_dynamic_modifier = { modifier = war_of_gazala_modifier scope = CAN }
					add_dynamic_modifier = { modifier = war_of_gazala_modifier scope = RAJ }
					add_dynamic_modifier = { modifier = war_of_gazala_modifier scope = AST }
					add_dynamic_modifier = { modifier = war_of_gazala_modifier scope = MEX }
					add_dynamic_modifier = { modifier = war_of_gazala_speed_modifier }
				}

				# THE FIX: test for >= 0 (not exact = 0) so any fixed-point overshoot
				# still triggers cleanup and the attack penalty can never get stranded
				# as a positive buff. NB: HOI4 check_variable has no ">=" shorthand
				# (vanilla uses zero of them), so use the explicit compare form.
				if = {
					limit = {
						check_variable = {
							var = global.war_of_gazala_modifier_attack_factor_var
							value = 0
							compare = greater_than_or_equals
						}
					}

					# Clamp to exactly 0 first so the modifiers read 0 (no buff blip)
					# during the frame they are removed.
					set_variable = { var = global.war_of_gazala_modifier_attack_factor_var value = 0 }
					set_variable = { var = global.war_of_gazala_modifier_speed_factor_var value = 0 }

					every_state = {
						limit = { has_state_flag = war_in_africa }
						# Mirror the scoped adds exactly so no per-tag instance (or
						# its icon) is left behind.
						remove_dynamic_modifier = { modifier = war_of_gazala_modifier scope = ITA }
						remove_dynamic_modifier = { modifier = war_of_gazala_modifier scope = GER }
						remove_dynamic_modifier = { modifier = war_of_gazala_modifier scope = HUN }
						remove_dynamic_modifier = { modifier = war_of_gazala_modifier scope = BUL }
						remove_dynamic_modifier = { modifier = war_of_gazala_modifier scope = ROM }
						remove_dynamic_modifier = { modifier = war_of_gazala_modifier scope = FIN }
						remove_dynamic_modifier = { modifier = war_of_gazala_modifier scope = SPR }
						remove_dynamic_modifier = { modifier = war_of_gazala_modifier scope = ENG }
						remove_dynamic_modifier = { modifier = war_of_gazala_modifier scope = FRA }
						remove_dynamic_modifier = { modifier = war_of_gazala_modifier scope = SAF }
						remove_dynamic_modifier = { modifier = war_of_gazala_modifier scope = CAN }
						remove_dynamic_modifier = { modifier = war_of_gazala_modifier scope = RAJ }
						remove_dynamic_modifier = { modifier = war_of_gazala_modifier scope = AST }
						remove_dynamic_modifier = { modifier = war_of_gazala_modifier scope = MEX }
						remove_dynamic_modifier = { modifier = war_of_gazala_speed_modifier }
						clr_state_flag = war_in_africa
					}

					clr_global_flag = war_of_gazala_active
					clear_variable = global.war_of_gazala_modifier_attack_factor_var
					clear_variable = global.war_of_gazala_modifier_speed_factor_var
				}
			}
		}
	}
	on_daily_WST = {
		effect = {
			# ---------------------------------------------------------------
			# ACTIVATION (one-shot, daily). Fires the first DAY Italy is at war with
			# the UK or France = the desert war has begun. war_of_gazala_fired is
			# permanent so it can never re-trigger; works in every game mode,
			# including 1941 saves where the war already exists at load.
			# Gated on the war_of_gazala game rule (default On; set Off in the
			# lobby to disable the whole mechanic - see common/game_rules/).
			# ---------------------------------------------------------------
			if = {
				limit = {
					has_game_rule = { rule = war_of_gazala option = on }
					NOT = { has_global_flag = war_of_gazala_fired }
					ITA = {
						OR = {
							has_war_with = ENG
							has_war_with = FRA
						}
					}
				}
				set_global_flag = war_of_gazala_fired
				set_global_flag = war_of_gazala_active
				# Skip token: the next monthly decay tick is a no-op (see decay block).
				set_variable = { var = global.war_of_gazala_skip_first_decay value = 1 }

				# Start at -50% attack / -25% speed (set BEFORE adding modifiers
				# so the first frame already reads the right value).
				set_variable = { var = global.war_of_gazala_modifier_attack_factor_var value = -0.5 }
				set_variable = { var = global.war_of_gazala_modifier_speed_factor_var value = -0.25 }

				every_state = {
					limit = {
						OR = {
							# --- Italian Libya ---
							state = 448 # Tripoli
							state = 449 # Libyan Coast
							state = 450 # Benghasi
							state = 451 # Derna
							state = 661 # Tripolitania
							state = 662 # Sirte
							state = 663 # Cyrenaica
							# --- British Egypt ---
							state = 446 # Cairo
							state = 447 # Alexandria
							state = 452 # Marsa Matruh
							state = 456 # Luxor
							state = 457 # Aswan
							state = 552 # Western Egypt (the Western Desert battlefield)
							state = 907 # Cairo (region)
							# --- French Tunisia ---
							state = 458 # Tunisia
							state = 665 # Gabes
							# --- French Algeria ---
							state = 459 # Western Algeria
							state = 460 # Eastern Algeria
							state = 513 # Southern Algeria
							# --- Anglo-Egyptian Sudan (East African approaches) ---
							state = 549 # Sudan
							state = 551 # Khartoum
							state = 883 # Kassala
							# (East Africa removed from the theatre 2026-08-08:
							# Ethiopia 271/835-843/908, Eritrea 550, Italian/French/
							# British Somaliland 559/268/269, Garissa 903 no longer
							# take the debuff.)
						}
					}
					set_state_flag = war_in_africa

					# Attack penalty - one instance per belligerent so BOTH the
					# Axis AND the Allies suffer it in the theatre.
					add_dynamic_modifier = { modifier = war_of_gazala_modifier scope = ITA } # Axis
					add_dynamic_modifier = { modifier = war_of_gazala_modifier scope = GER }
					add_dynamic_modifier = { modifier = war_of_gazala_modifier scope = HUN }
					add_dynamic_modifier = { modifier = war_of_gazala_modifier scope = BUL }
					add_dynamic_modifier = { modifier = war_of_gazala_modifier scope = ROM }
					add_dynamic_modifier = { modifier = war_of_gazala_modifier scope = FIN }
					add_dynamic_modifier = { modifier = war_of_gazala_modifier scope = SPR }
					add_dynamic_modifier = { modifier = war_of_gazala_modifier scope = ENG } # Allies
					add_dynamic_modifier = { modifier = war_of_gazala_modifier scope = FRA }
					add_dynamic_modifier = { modifier = war_of_gazala_modifier scope = SAF }
					add_dynamic_modifier = { modifier = war_of_gazala_modifier scope = CAN }
					add_dynamic_modifier = { modifier = war_of_gazala_modifier scope = RAJ }
					add_dynamic_modifier = { modifier = war_of_gazala_modifier scope = AST }
					add_dynamic_modifier = { modifier = war_of_gazala_modifier scope = MEX }

					# Speed penalty - state-wide (slows everyone in the desert).
					add_dynamic_modifier = { modifier = war_of_gazala_speed_modifier }
				}
			}
		}
	}
}
