on_actions = {
	on_startup = {
		effect = {
			if = {
				limit = { NOT = { has_global_flag = bookmark_on_startup_done } }
				set_global_flag = bookmark_on_startup_done

				if = {
					limit = { GER = { has_country_flag = barbstart } }
					if = {
						limit = {
							date < 1941.1.21
						}
						random_country = { news_event = { id = barbmode.1 } }
					}
					every_country = {
						limit = {
							is_playable_country = yes
						}
						add_ideas = tradeback_is_cancer
					}
					GER = {
						add_offsite_building = {
							type = industrial_complex
							level = 118
						}
						add_offsite_building = {
							type = arms_factory
							level = 99
						}
						add_political_power = 999
						army_experience = 999
						navy_experience = 999
						air_experience = 999
					}

					SOV = {
						add_offsite_building = {
							type = arms_factory
							level = 180
						}
						add_political_power = 999
						army_experience = 999
						navy_experience = 999
						air_experience = 999
					}

					ROM = {
						add_political_power = 999
						army_experience = 999
						navy_experience = 999
						air_experience = 999
					}

					# State history parser rejects building levels > 15
					# ("Malformed token"); these states' 1941 blocks are
					# clamped to 15 and topped up to intended levels here
					219 = { # Moscow
						set_building_level = { type = arms_factory level = 20 instant_build = yes }
					}
					43 = { # Hungary (ROM in barb)
						set_building_level = { type = arms_factory level = 20 instant_build = yes }
					}
					46 = { # Muntenia
						set_building_level = { type = arms_factory level = 16 instant_build = yes }
					}
				}

				# Consolidation mode (reworked No-Asia): merged minors' armies and
				# assets transfer to their union partner at game start
				if = {
					limit = { GER = { has_country_flag = stb_asia_bookmark } }
					if = {
						limit = {
							SAF = { exists = yes }
							NOT = { country_exists = CAN }
						}
						SAF = {
							add_equipment_to_stockpile = { type = convoy_1 amount = 125 }
							if = {
								limit = { has_dlc = "No Step Back" }
								load_oob = "CAN_1936_nsb_union"
								else = { load_oob = "CAN_1936_union" }
							}
						}
					}
					if = {
						limit = {
							AST = { exists = yes }
							NOT = { country_exists = RAJ }
						}
						AST = {
							add_equipment_to_stockpile = { type = convoy_1 amount = 20 }
							if = {
								limit = { has_dlc = "Graveyard of Empires" }
								load_oob = "RAJ_1936_GOE"
								else = { load_oob = "RAJ_1936" }
							}
						}
					}
					if = {
						limit = {
							AST = { exists = yes }
							NOT = { country_exists = MEX }
						}
						AST = {
							add_equipment_to_stockpile = { type = convoy_1 amount = 45 }
							load_oob = "MEX_1936"
						}
					}
					if = {
						limit = {
							ROM = { exists = yes }
							NOT = { country_exists = HUN }
						}
						ROM = {
							add_equipment_to_stockpile = { type = convoy_1 amount = 50 }
							if = {
								limit = { has_dlc = "Gotterdammerung" }
								load_oob = "HUN_wuw_1936_union"
								else = { load_oob = "HUN_1936" }
							}
						}
					}
					if = {
						limit = {
							SPR = { exists = yes }
							NOT = { country_exists = FIN }
						}
						SPR = {
							add_ideas = extra_cp
							add_equipment_to_stockpile = { type = convoy_1 amount = 5 }
							load_oob = "FIN_1936_union"
						}
					}
				}

				if = {
					limit = {
						NOT = { country_exists = USA }
					}
					ENG = {
						add_research_slot = 1

						country_event = {
							id = msb.4
							days = 1825
						}

					}

					WST = {
						give_resource_rights = {
							receiver = ENG
							state = 378
						}
						give_resource_rights = {
							receiver = ENG
							state = 375
						}
						give_resource_rights = {
							receiver = ENG
							state = 374
						}
						country_event = {
							id = msb.3
							days = 560
						}
					}
				}
			}
		}
	}
}
