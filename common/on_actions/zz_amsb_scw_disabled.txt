# AMSB 2026-08-15: "Spanish Civil War" lobby rule (common/game_rules/00_game_rules.txt: amsb_spanish_civil_war).
# Default = disabled: Spain starts as Nationalist Spain under Franco and the civil war never fires.
# The world is put into the exact post-war state every other script already understands
# (spanish_civil_war + scw_over + nationalist_victory), so:
#   - the plot missions in common/decisions/SPR.txt never activate (they need NOT spanish_civil_war),
#   - lar_spain.2 is never fired,
#   - the post-war Nationalist tree unlocks (focuses gated on scw_over),
#   - msb on_monthly_WST logic (no_faction lift) is satisfied on day one,
#   - AI strategies for SPR/POR/SOV take their post-SCW branches.
# Runs AFTER 05_lar_on_actions.txt (which locks SPR templates) and msb_on_actions.txt (which adds
# no_faction) because on_action files execute in file-name order; both are undone here.
# Guarded by date so it never re-fires on a later bookmark and cannot double-apply.
on_actions = {
	on_startup = {
		effect = {
			if = {
				limit = {
					date < 1936.1.2
					has_game_rule = { rule = amsb_spanish_civil_war option = disabled }
					SPR = { exists = yes }
					NOT = { has_global_flag = spanish_civil_war }
				}
				set_global_flag = spanish_civil_war
				set_global_flag = scw_over
				set_global_flag = nationalist_victory
				SPR = {
					set_country_flag = SPR_nationalist_spain_flag
					set_cosmetic_tag = SPR_nationalist_spain
					country_lock_all_division_template = no
					# Republic-era spirits (same removals as the Franco option of lar_spain.2)
					remove_ideas = {
						SPR_military_disloyalty
						SPR_political_violence
					}
					SPR_clr_national_strikes = yes
					if = {
						limit = { has_idea = no_faction }
						remove_ideas = no_faction
					}
					set_rule = { can_join_factions = yes }
					set_politics = {
						ruling_party = neutrality
						elections_allowed = no
					}
					set_popularities = {
						neutrality = 60
						fascism = 20
						democratic = 15
						communism = 5
					}
					set_party_name = {
						ideology = neutrality
						long_name = SPR_fascism_francoist_party_long
						name = SPR_fascism_francoist_party
					}
					kill_country_leader = yes
					kill_ideology_leader = fascism
					add_country_leader_role = {
						character = SPA_francisco_franco
						country_leader = {
							ideology = despotism
							traits = { caudillo_1 hardened_officer generallissimo }
							expire = "1965.1.1"
						}
						promote_leader = yes
					}
					# fascist role too (mirrors SPA_caudillo_of_spain) so a later fascist switch keeps Franco
					add_country_leader_role = {
						character = SPA_francisco_franco
						country_leader = {
							ideology = fascism_ideology
							traits = { caudillo_1 hardened_officer generallissimo }
							expire = "1965.1.1"
						}
						promote_leader = no
					}
					# Pre-war plot branch counted as done WITHOUT its rewards (no plot machinery, no units).
					# SPA_unify_the_nationalist_front is deliberately NOT unlocked here: its own
					# available block is always = no and every other focus chains from it, so the
					# whole tree stays locked until amsb_scw.3 unlocks it on 1938-01-01 (user choice:
					# hard lock, no head start on the post-war tree).
					unlock_national_focus = SPA_a_great_spain
					unlock_national_focus = SPA_hand_over_the_ceda_campaign_chest
					unlock_national_focus = SPA_negotiate_carlist_support
					unlock_national_focus = SPA_the_army_of_africa
					unlock_national_focus = SPA_secure_the_northern_garrisons
					unlock_national_focus = SPA_con_paquito
					# 1936.1.1 + 731 days = 1938.1.1 (1936 is a leap year)
					country_event = { id = amsb_scw.3 days = 731 }
				}
				# 1936.1.1 + 213 days = 1936.8.1: +10% war support for everyone (events/amsb_scw_disabled.txt).
				# Carried on WST (always-extant utility tag) so it fires exactly once.
				WST = {
					country_event = { id = amsb_scw.1 days = 213 }
				}
			}
		}
	}
}
