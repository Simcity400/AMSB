# AMSB 2026-08-02: array hygiene for annexed countries.
#
# Evidence (full dump hoi4.exe.137352.dmp, crash 2026-08-01 20:22:09, LFH bitmap
# corruption): the corrupted heap bucket is dense with CDecisionStatus tracking
# items (SDecisionRandomCountItem -> CDecision) and relation-strip country-flag
# UI items (CStripFlagItem@CRelationStripViewBase), several holding pointers
# into decommitted memory. Vanilla on_declare_war pushes raw country scopes
# into GER.GER_war_targets_array and NOTHING ever removes them; two GER
# targeted decisions dereference every element continuously. That session:
# Netherlands entered the array 20:14:03 and was annexed 20:14:41.
#
# Mitigation: drop a country from the array the moment it is annexed.
# on_annex: ROOT = annexing winner, FROM = country being annexed (vanilla doc).
on_actions = {
	on_annex = {
		effect = {
			if = {
				limit = {
					GER = { check_variable = { GER_war_targets_array^num > 0 } }
				}
				GER = {
					remove_from_array = {
						array = GER_war_targets_array
						value = FROM
					}
				}
			}
		}
	}
}
