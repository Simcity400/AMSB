AMSB_finland_no_naval_inv_protected_state_effect = {
	if = {
		limit = {
			OWNER = { NOT = { tag = SOV } }
		}
		set_state_flag = finnish_demilitarized_zone
		if = {
			limit = {
				OWNER = {
					OR = {
						tag = ROM
						tag = HUN
						tag = SPR
						tag = BUL
					}
				}
			}
			set_demilitarized_zone = yes
		}
	}
}

AMSB_finland_no_naval_inv_soviet_border_state_effect = {
	if = {
		limit = {
			OWNER = { tag = SOV }
		}
		clr_state_flag = finnish_demilitarized_zone
		set_demilitarized_zone = no
	}
}

AMSB_spain_no_naval_inv_protected_state_effect = {
	set_state_flag = spain_demilitarized_zone
	if = {
		limit = {
			OWNER = {
				OR = {
					tag = ROM
					tag = HUN
					tag = SPR
					tag = BUL
				}
			}
		}
		set_demilitarized_zone = yes
	}
}
