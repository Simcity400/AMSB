hmm_capitulations_on_actions = {
	# Poland
	if = {
		limit = {
			original_tag = POL
			POL = { has_war_with = GER }
		}
		ENG = {
			country_event = { id = britain.9 days = 1 }
		}
		SOV = {
			transfer_state = 784
			transfer_state = 96
			transfer_state = 95
			transfer_state = 94
			transfer_state = 93
			transfer_state = 91
			transfer_state = 89
			transfer_state = 97
		}
		
	}

	# Netherlands
	if = {
		limit = {
			original_tag = HOL
			GER = { has_war_with = HOL }
			NOT = {
				JAP = {
					has_war_with = HOL
				}
			}
		}
		HOL = {
			add_stability = 1.0
			add_equipment_to_stockpile = {
                type = convoy_1
                amount = 1000
            }
			add_equipment_to_stockpile = {
				type = infantry_equipment_0
				amount = 20000
				producer = HOL
			}
		}

		ENG = { # HOL STATES GO TO UK POST CAP
			# AMSB 2026-06-13: removed the Indonesian state transfers (672/334/335/667/673/668).
			# They date from the pre-1.19 tree where the East Indies were HOL-owned states; since
			# the port they belong to independent INS, and transfer_state ignores ownership - this
			# was ripping Indonesia from INS to ENG on Dutch capitulation (user-reported bug).
			# ENG keeps only genuinely Dutch territory: Suriname + the HOL home states via annex.
			transfer_state = 309

			annex_country = {
				target = HOL
				transfer_troops = no
			}
		}
	}

	if = {
		limit = {
			original_tag = GRE
			ITA = { has_war_with = GRE }
		}
		
		ENG = {
			annex_country = {
				target = GRE
				transfer_troops = no
			}
		}
	}

	# China
	if = {
		limit = { 
			original_tag = CHI
			JAP = { has_war_with = CHI }
		} 
		JAP = {
			add_named_threat = { 
				threat = 30
				name = hmm_chi
			}
			annex_country = {
				target = CHI
				transfer_troops = no
			}
		}
	}

	# Japan
	if = {
		limit = { 
			original_tag = JAP
			USA = { has_war_with = JAP }
		} 
		JAP = {
			transfer_navy = { target = USA }
		}
		USA = {
			annex_country = {
				target = JAP
				transfer_troops = no
			}
		}
	}

	# Soviet Union
	if = {
		limit = { 
			original_tag = SOV
			GER = { has_war_with = SOV }
		} 
		SOV = {
			transfer_navy = { target = GER }
		}
		GER = {
			annex_country = {
				target = SOV
				transfer_troops = no
			}
		}
		if = { 
			limit= { country_exists = ROM }  
			ROM = {
				transfer_state = 766
				transfer_state = 78
				transfer_state = 80
			}
		}
		if = { 
			limit= { country_exists = FIN }  
			FIN = {
				transfer_state = 213
				transfer_state = 215
				transfer_state = 216
				transfer_state = 146
			}
		}
	}
	if = {
		limit = { 
			OR = {
				tag = BEL
				tag = HOL
				tag = POL
				tag = NOR
				tag = DEN
				tag = LUX
			}
		} 
		ROOT = {
			remove_ideas = {
				ai_war
			}
			delete_unit = {}
		}
		
	}
}
