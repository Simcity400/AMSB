# =============================================================================
# WAR OF GAZALA - desert-war attrition modifiers (restored & fixed 2026-06-21)
# =============================================================================
# Both modifiers are STATE modifiers driven by a GLOBAL variable, so a single
# monthly tick can ramp the whole African theatre back to 0 at once.
# The driving logic lives in common/on_actions/gazala_on_actions.txt.
#
# Attack penalty is applied per-belligerent (scope = TAG) so BOTH sides feel
# the desert; speed penalty is applied state-wide. Values are negative while
# the desert war is "fresh" and decay linearly to 0, at which point the
# on_actions logic removes the modifiers entirely.

war_of_gazala_modifier = {            # ATTACK penalty (added once per belligerent tag)
	enable = { always = yes }

	icon = GFX_modifiers_SOV_armored_battery

	army_attack_factor = global.war_of_gazala_modifier_attack_factor_var
}

war_of_gazala_speed_modifier = {      # SPEED penalty (state-wide)
	enable = { always = yes }

	icon = GFX_modifiers_SOV_armored_battery

	army_speed_factor = global.war_of_gazala_modifier_speed_factor_var
}
