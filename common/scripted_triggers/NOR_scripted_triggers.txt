###### AI Triggers ##### 

NOR_is_historical_ai = { 
	is_ai = yes 
	OR = {
		AND = {
			is_historical_focus_on = yes 
			has_game_rule = {
				rule = NOR_AI_BEHAVIOR
				option = DEFAULT
			}
		}
		has_country_flag = NOR_AI_RANDOM_HISTORICAL
		has_game_rule = {
			rule = NOR_AI_BEHAVIOR
			option = HISTORICAL
		}
	}
}
