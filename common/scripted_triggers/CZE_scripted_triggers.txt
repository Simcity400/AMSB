    all_controlled_state = {
        OR = {
            state = FROM
            NOT = {
                has_dynamic_modifier = { 
                    modifier = CZE_sudetenland_german_state_modifier
                    scope = ROOT
                }
                has_dynamic_modifier = { 
                    modifier = CZE_czech_state_modifier
                    scope = ROOT

                    modifier = CZE_tesinko_polish_state_modifier
